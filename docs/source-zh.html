710local function scan_step(L)
711 local t_end = clock() + SLICE_S
712 while clock() < t_end do
713 if L.rbase == nil then
714 if L.addr >= USER_END or k32.VirtualQuery(ffi.cast('const void*', L.addr), mbi, MBI_SIZE) == 0 then
715 return true
716 end
717 local m = mbi[0]
718 local rb, rs = tonumber(ffi.cast('uintptr_t', m.BaseAddress)), tonumber(m.RegionSize)
719 if rb + rs <= L.addr then return true end
720 L.addr = rb + rs
721 if m.State == MEM_COMMIT and m.Type == MEM_PRIVATE and m.Protect == PAGE_READWRITE then
722 L.rbase, L.rend, L.off = rb, rb + rs, 0
723 end
724 else
725 local take = math.min(CHUNK, L.rend - L.rbase - L.off)
726 local cbase = L.rbase + L.off
727 local got = safe_read(cbase, take, SCRATCH)
728 -- Compare aligned int32s in place: building a Lua string would leave stale copies of the
729 -- record in the heap for later scans to find.
730 local ok = got == take
731 if ok then
732 local words = ffi.cast('const int32_t*', SCRATCH)
733 for i = 0, got / 4 - 1 do
734 if words[i] == 6 and words[i + 1] == 10 and i + 1 < got / 4 then
735 test_anchor(L, cbase + i * 4 - OFF_ARMOR0)
736 end
737 end
738 end
739 L.off = L.off + take
740 L.scanned = L.scanned + take
741 if not ok or L.off >= L.rend - L.rbase then L.rbase = nil end
742 end
743 end
744 return false
745end

1300-- Superellipse ("squircle") centred at cx, cy with half-sizes a, b; n = 2 ellipse, larger = boxier.
1301local function squircle(cx, cy, a, b, n, k)
1302 local pts = {}
1303 k = k or 28

1306 local c, s = math.cos(t), math.sin(t)

1418-- Pixel-art rasteriser: a global PIX-sized grid, a cell is filled when its centre
1419-- lies inside any triangle; filled cells are merged into horizontal runs.
1420local PIX = 4
1421local function inside(t, x, y)
1422 local function side(a, b) return (b[1] - a[1]) * (y - a[2]) - (b[2] - a[2]) * (x - a[1]) end
1423 local d1, d2, d3 = side(t[1], t[2]), side(t[2], t[3]), side(t[3], t[1])
1424 local neg = d1 < 0 or d2 < 0 or d3 < 0
1425 local pos = d1 > 0 or d2 > 0 or d3 > 0
1426 return not (neg and pos)
1427end
1428local function pixelate(tris)
1429 local minx, maxx, miny, maxy = 1e9, -1e9, 1e9, -1e9
1430 for _, t in ipairs(tris) do
1431 for i = 1, 3 do
1432 minx, maxx = math.min(minx, t[i][1]), math.max(maxx, t[i][1])
1433 miny, maxy = math.min(miny, t[i][2]), math.max(maxy, t[i][2])
1434 end
1435 end
1436 local out = {}
1437 for gy = math.floor(miny / PIX), math.ceil(maxy / PIX) do
1438 local run
1439 for gx = math.floor(minx / PIX), math.ceil(maxx / PIX) do
1440 local cx, cy = (gx + 0.5) * PIX, (gy + 0.5) * PIX
1441 local hit = false
1442 for _, t in ipairs(tris) do if inside(t, cx, cy) then hit = true; break end end
1443 if hit then
1444 if run then run[3] = run[3] + PIX else run = {gx * PIX, gy * PIX, PIX, PIX} end
1445 elseif run then out[#out + 1] = run; run = nil end
1446 end
1447 if run then out[#out + 1] = run end
1448 end
1449 return out

1456-- ---------- poly style ----------
1457-- Low-poly look: every part is 3-5 big flat facets (a few raised vertices make them face different ways), each lit
1458-- flat from the top-left; shades are quantised so neighbouring facets of one tone merge.
1459-- Each facet is one Gui.triangle. Static RE of helldivers2.exe: the Gui.triangle binding
1460-- (0x3E12C0) takes (gui, p1, p2, p3, layer, [color], [material], [uv x3]) with screen x = v.x, screen y = v.z
1461-- (v.y = depth), and without a material it queues the same command as Gui.rect (0x267990, material id 0xFFFFFF =
1462-- the gui default; Gui.rect is itself 2 triangles there). Fallback: horizontal spans POLY_STEP design px tall.
1463-- H.POLY parts: {slot, label, facets = {{tri, shade}, ...}}
1464H.POLY = {}
1465do
1466 local STEP_P = 2
1467 H.POLY_STEP = STEP_P
1468 local LX, LY, LZ = -0.45, -0.6, 0.66
1469 local ll = math.sqrt(LX * LX + LY * LY + LZ * LZ)
1470 LX, LY, LZ = LX / ll, LY / ll, LZ / ll
1471 local function lit(a, b, c)
1472 local ux, uy, uz = b[1] - a[1], b[2] - a[2], (b[3] or 0) - (a[3] or 0)
1473 local vx, vy, vz = c[1] - a[1], c[2] - a[2], (c[3] or 0) - (a[3] or 0)
1474 local nx, ny, nz = uy * vz - uz * vy, uz * vx - ux * vz, ux * vy - uy * vx
1475 local n = math.sqrt(nx * nx + ny * ny + nz * nz)
1476 if n == 0 then return 1 end
1477 if nz < 0 then nx, ny, nz = -nx, -ny, -nz end
1478 local d = math.max(0, (nx * LX + ny * LY + nz * LZ) / n)
1479 return math.min(1, math.floor((0.42 + 0.58 * d) * 25 + 0.5) / 25)
1480 end

1543 -- x-interval of a polygon (or triangle) at height y, nil when the line misses it.
1544 local function span(pts, y)
1545 local lo, hi
1546 for i = 1, #pts do
1547 local a, b = pts[i], pts[i % #pts + 1]
1548 if (a[2] - y) * (b[2] - y) <= 0 and a[2] ~= b[2] then
1549 local x = a[1] + (y - a[2]) * (b[1] - a[1]) / (b[2] - a[2])
1550 lo, hi = math.min(lo or x, x), math.max(hi or x, x)
1551 end
1552 end
1553 return lo, hi
1554 end

233 -- means the entry now belongs to another unit (the table was compacted): look it up again, write nothing.
234 local sane = hi and bitlib.band(hi, 3) == 3
235 if v and not sane then L.vis = nil end

237 local hp = i32(Z_OFF + slot * 4)

240 if M.enabled and sane and died and not (f and f.gave_up) then
241 local applied_hidden = alo and mbit(alo, ahi, pm)
242 local shown = not mbit(lo, hi, pm) and not applied_hidden
243 if shown and t - died >= FIX_DELAY then
244 f = f or {n = 0}
245 L.fix[slot] = f
246 if t >= (f.next or 0) then
247 if f.n >= FIX_TRIES then
248 f.gave_up = true

250 L.ent, slot, f.n, M.masks(v)))
251 else

253 local ok = set_bit(v.override, pm) and set_bit(v.touched, pm)
254 writer(string.format('entity 0x%X plate slot %d: dead %.1f s but still drawn -> hid model mask %d (%s), try %d (%s)',
255 L.ent, slot, t - died, pm, ok and 'ok' or 'write failed', f.n, M.masks(v)))
256 end
257 end
258 elseif not shown and f and f.at and not f.done then
259 f.done = true
260 writer(string.format('entity 0x%X plate slot %d: hidden %.1f s after the fix, in the %s mask (%s)', L.ent, slot,
261 t - f.at, applied_hidden and 'applied' or 'base', M.masks(v)))
262 end
263 end
264 end
